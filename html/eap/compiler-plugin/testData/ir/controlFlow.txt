import androidx.compose.runtime.Composable
import org.jetbrains.compose.web.dom.*

@Composable
fun ControlFlow(values: List<String>, show: Boolean) {
    Div {
        if (show) {
            Span { Text("heading") }
        }
        for (value in values) {
            if (value.isEmpty()) continue
            Span { Text(value) }
        }
        Text("end")
    }
}
