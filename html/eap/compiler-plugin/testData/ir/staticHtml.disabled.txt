import androidx.compose.runtime.Composable
import org.jetbrains.compose.web.dom.*

private const val CARD = "card"

@Composable
fun StaticHtml() {
    Div(attrs = {
        classes(CARD)
        attr("title", "A & \"B\"")
    }) {
        Span { Text("<&" + "Hello>") }
        Br()
    }
}


// Transformed IR: main
// ------------------------------------------
// MODULE: <main>
// FILE: staticHtml.kt

private const val CARD: String
  field = "card"
  private get

@Composable
fun StaticHtml() {
  Div(attrs = local fun AttrsScope<HTMLDivElement>.<anonymous>() {
    $this$Div.classes(classes = ["card"])
    $this$Div.attr(attr = "title", value = "A & \"B\"") /*~> Unit */
  }
, content = local fun ElementScope<HTMLDivElement>.<anonymous>() {
    Span(content = local fun ElementScope<HTMLSpanElement>.<anonymous>() {
      Text(value = "<&Hello>")
    }
)
    Br()
  }
)
}
